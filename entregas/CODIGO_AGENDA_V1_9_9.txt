Public Sub AgendaSISDENF_Abrir(ByVal frm As Access.Form)
    mQ = "": mData = "": mCat = "": mTipo = "todos": mPeriodo = "": mPage = 1: mAviso = "": mAvisoErro = False
    mAtualizacaoPendente = False: mAtualizacaoEmCurso = False: mArquivoIndice = 0
    AgendaSISDENF_Atualizar frm
End Sub

Public Sub AgendaSISDENF_Atualizar(ByVal frm As Access.Form)
    On Error GoTo Trata
    Dim caminho As String, urlArquivo As String
    Dim navegador As Object
    If mBusy Or mAtualizacaoEmCurso Then
        mAtualizacaoPendente = True
        Exit Sub
    End If
    mBusy = True
    mAtualizacaoEmCurso = True
    caminho = AgendaSISDENF_GerarHTML()
    If Len(caminho) = 0 Then Err.Raise vbObjectError + 1930, , "Geracao do HTML retornou caminho vazio."
    If Len(Dir$(caminho)) = 0 Then Err.Raise vbObjectError + 1931, , "Arquivo HTML nao encontrado: " & caminho
    urlArquivo = "file:///" & Replace(caminho, Chr$(92), "/")
    Set navegador = frm.Controls("WebBrowser1").Object
    navegador.Navigate urlArquivo
    mAtualizacaoPendente = False
Saida:
    mAtualizacaoEmCurso = False
    mBusy = False
    Set navegador = Nothing
    Exit Sub
Trata:
    mAtualizacaoPendente = False
    Debug.Print "Agenda SISDENF falha Navigate/GerarHTML: " & Err.Number & " - " & Err.Description
    MsgBox "Agenda - falha ao atualizar: " & Err.Number & " - " & Err.Description, vbExclamation
    Resume Saida
End Sub

Public Sub AgendaSISDENF_VerificarComando(ByVal frm As Access.Form)
    ' Processar no Timer: nao navegar novamente durante evento do navegador.
    On Error GoTo Falha
    Dim doc As Object, ponte As Object, comando As String, cancelar As Boolean
    If mBusy Or mAtualizacaoEmCurso Then Exit Sub
    If mAtualizacaoPendente Then
        mAtualizacaoPendente = False
        AgendaSISDENF_Atualizar frm
        Exit Sub
    End If
    Set doc = frm.Controls("WebBrowser1").Object.Document
    If doc Is Nothing Then Exit Sub
    Set ponte = doc.getElementById("agBridge")
    If ponte Is Nothing Then Exit Sub
    comando = CStr(Nz(ponte.getAttribute("data-command"), ""))
    If Len(comando) = 0 Then Exit Sub
    ponte.setAttribute "data-command", ""
    cancelar = False
    AgendaSISDENF_Navegar frm, "about:blank?" & comando, cancelar
    Exit Sub
Falha:
    Debug.Print "Agenda SISDENF ponte: " & Err.Number & " - " & Err.Description
End Sub

Private Sub AgSolicitarAtualizacao()
    mAtualizacaoPendente = True
End Sub

Public Sub AgendaSISDENF_Navegar(ByVal frm As Access.Form, ByVal Endereco As String, ByRef Cancel As Boolean)
    On Error GoTo Falha
    Dim p As Long, s As String, acao As String
    If InStr(1, Endereco, "about:blank?", vbTextCompare) <> 1 Then Exit Sub
    p = InStr(1, Endereco, "?", vbBinaryCompare)
    s = Mid$(Endereco, p + 1)
    acao = AgGet(s, "agcmd")
    If Len(acao) = 0 Then Exit Sub
    Cancel = True
    ' Excluir somente por ID valido, com confirmacao e transacao DAO.
    If acao = "delete" Then
        Dim idExcluir As Long
        idExcluir = AgIDValido(AgGet(s, "id"))
        If AgExcluirCompromisso(idExcluir) Then
            mAviso = "Compromisso ID " & CStr(idExcluir) & " excluido com sucesso."
            mAvisoErro = False
            AgSolicitarAtualizacao
        End If
        Exit Sub
    End If
    If acao = "science" Then
        Cancel = True
        On Error GoTo Falha
        Dim idC As Long
        idC = AgIDValido(AgGet(s, "id"))
        Agenda18_ConfirmarCiencia idC, Environ$("USERNAME")
        mAviso = "Ciência registrada para o compromisso ID " & CStr(idC) & "."
        mAvisoErro = False
        AgSolicitarAtualizacao
        Exit Sub
    End If
    If acao = "category" Then
        Cancel = True
        Agenda18_CadastrarCategoria AgGet(s, "code"), AgGet(s, "name"), AgGet(s, "color")
        mAviso = "Categoria cadastrada com sucesso."
        mAvisoErro = False
        AgSolicitarAtualizacao
        Exit Sub
    End If
    If acao = "editcategory" Then
        Cancel = True
        AgEditarCategoria AgGet(s, "code"), AgGet(s, "name"), _
                          AgGet(s, "color"), AgGet(s, "active")
        mAviso = "Categoria atualizada com sucesso."
        mAvisoErro = False
        AgSolicitarAtualizacao
        Exit Sub
    End If
    If acao = "save" Then
        Cancel = True
        If AgendaSISDENF_Gravar(s) Then
            mPage = 1: mQ = "": mData = AgGet(s, "data"): mCat = "": mTipo = "todos": mPeriodo = ""
        End If
        AgSolicitarAtualizacao
        Exit Sub
    End If
    If acao <> "search" Then Err.Raise vbObjectError + 1954, , "Comando desconhecido."
    Cancel = True
    mQ = AgGet(s, "q"): mData = AgGet(s, "dt")
    mCat = AgGet(s, "cat"): mTipo = AgGet(s, "tipo"): mPeriodo = AgGet(s, "periodo")
    If mTipo <> "lidos" And mTipo <> "pendentes" Then mTipo = "todos"
    If Len(mCat) <> 2 Or Not (mCat Like "##") Then mCat = ""
    Dim dataPeriodo As Date
    If Len(mPeriodo) > 0 And mPeriodo <> "10" Then
        If Not AgParseDate(mPeriodo, dataPeriodo) Then Err.Raise vbObjectError + 1956, , "Periodo invalido."
    End If
    Dim paginaTexto As String
    paginaTexto = AgGet(s, "page")
    If Len(paginaTexto) = 0 Then paginaTexto = "1"
    If Len(paginaTexto) > 6 Or Not (paginaTexto Like String$(Len(paginaTexto), "#")) Then _
        Err.Raise vbObjectError + 1955, , "Pagina invalida."
    mPage = CLng(paginaTexto)
    If mPage < 1 Then mPage = 1
    If mPage > 100000 Then mPage = 100000
    AgSolicitarAtualizacao
    Exit Sub
Falha:
    Cancel = True
    mAviso = "OPERACAO NAO CONCLUIDA: " & Err.Description
    mAvisoErro = True
    AgSolicitarAtualizacao
End Sub

Private Function AgExcluirCompromisso(ByVal idEvento As Long) As Boolean
    ' Nao exclui sem mostrar ao usuario o registro real encontrado no banco.
    ' A transacao reverte ambas as tabelas se existir alguma restricao.
    On Error GoTo Falha
    Dim db As DAO.Database, ws As DAO.Workspace, rs As DAO.Recordset
    Dim assunto As String, dataTexto As String
    Dim emTransacao As Boolean, qtdExcluida As Long
    If idEvento <= 0 Then Err.Raise vbObjectError + 1952, , "ID de compromisso invalido."
    Set db = CurrentDb
    Set rs = db.OpenRecordset("SELECT [ID],[Agenda],[Data] FROM [T_Agendamento] WHERE [ID]=" & CStr(idEvento), dbOpenSnapshot)
    If rs.EOF Then
        mAviso = "O compromisso ID " & CStr(idEvento) & " nao existe mais. Atualize a lista."
        mAvisoErro = True
        AgSolicitarAtualizacao
        GoTo Saida
    End If
    assunto = Left$(Nz(rs.Fields("Agenda").Value, ""), 180)
    dataTexto = AgDate(rs.Fields("Data").Value)
    rs.Close: Set rs = Nothing
    If MsgBox("EXCLUIR DEFINITIVAMENTE O COMPROMISSO?" & vbCrLf & vbCrLf & _
              "ID: " & CStr(idEvento) & vbCrLf & _
              "Data: " & dataTexto & vbCrLf & _
              "Assunto: " & assunto & vbCrLf & vbCrLf & _
              "As confirmacoes de ciencia deste evento tambem serao excluidas." & vbCrLf & _
              "Esta operacao nao pode ser desfeita.", _
              vbYesNo + vbQuestion + vbDefaultButton2, "Agenda SISDENF - Confirmar exclusao") <> vbYes Then
        GoTo Saida
    End If
    Set ws = DBEngine.Workspaces(0)
    ws.BeginTrans
    emTransacao = True
    db.Execute "DELETE FROM [T_AgendaCiencia] WHERE [IDAgendamento]=" & CStr(idEvento), dbFailOnError
    db.Execute "DELETE FROM [T_Agendamento] WHERE [ID]=" & CStr(idEvento), dbFailOnError
    qtdExcluida = db.RecordsAffected
    If qtdExcluida <> 1 Then Err.Raise vbObjectError + 1953, , "Exclusao nao confirmada (ID alterado ou removido por outro usuario)."
    ws.CommitTrans
    emTransacao = False
    AgExcluirCompromisso = True
Saida:
    On Error Resume Next
    If Not rs Is Nothing Then rs.Close
    Set rs = Nothing: Set db = Nothing: Set ws = Nothing
    Exit Function
Falha:
    Dim numeroErro As Long, textoErro As String
    numeroErro = Err.Number: textoErro = Err.Description
    On Error Resume Next
    If emTransacao Then ws.Rollback
    If Not rs Is Nothing Then rs.Close
    Set rs = Nothing: Set db = Nothing: Set ws = Nothing
    On Error GoTo 0
    Err.Raise numeroErro, "AgExcluirCompromisso", textoErro
End Function

Private Function AgendaSISDENF_Gravar(ByVal qs As String) As Boolean
    On Error GoTo Falha
    Dim db As DAO.Database, rs As DAO.Recordset, ver As DAO.Recordset
    Dim ws As DAO.Workspace, emTransacao As Boolean
    Dim assunto As String, categoria As String, dataTxt As String
    Dim horaTxt As String, fimTxt As String, dataEvento As Date
    Dim hInicio As Date, hFim As Variant, novoID As Long, termino As Date
    Dim dataFimTxt As String, dataFim As Date
    mAvisoErro = False
    assunto = Trim$(AgGet(qs, "assunto"))
    categoria = AgGet(qs, "categoria")
    dataTxt = AgGet(qs, "data")
    horaTxt = AgGet(qs, "hora")
    fimTxt = Trim$(AgGet(qs, "fim"))
    dataFimTxt = Trim$(AgGet(qs, "datafim"))
    If Len(assunto) = 0 Or Len(assunto) > 450 Then Err.Raise vbObjectError + 1001, , "Informe um assunto de ate 450 caracteres."
    If Len(categoria) <> 2 Or Not (categoria Like "##") Then Err.Raise vbObjectError + 1002, , "Categoria invalida."
    If Not Agenda18_CategoriaAtiva(categoria) Then Err.Raise vbObjectError + 1008, , "Categoria nao cadastrada ou inativa."
    If Not AgParseDate(dataTxt, dataEvento) Then Err.Raise vbObjectError + 1003, , "Data invalida. Use dd/mm/aaaa."
    If Len(dataFimTxt) > 0 Then
        If Not AgParseDate(dataFimTxt, dataFim) Then Err.Raise vbObjectError + 1009, , "Data final invalida."
        If dataFim < dataEvento Then Err.Raise vbObjectError + 1010, , "A data final nao pode ser anterior a inicial."
    End If
    If Not AgHoraValida(horaTxt, hInicio) Then Err.Raise vbObjectError + 1004, , "Hora inicial invalida. Use HH:MM."
    hFim = Null
    If Len(fimTxt) > 0 Then
        If Not AgHoraValida(fimTxt, termino) Then Err.Raise vbObjectError + 1005, , "Hora de termino invalida."
        If IIf(Len(dataFimTxt) > 0, dataFim, dataEvento) + termino < dataEvento + hInicio Then Err.Raise vbObjectError + 1006, , "Hora final anterior a inicial: revise os horarios."
        hFim = termino
    End If
    assunto = AgAssuntoExibicao(assunto)
    If Len(assunto) = 0 Then Err.Raise vbObjectError + 1001, , "Informe o assunto do compromisso."
    assunto = categoria & " - " & assunto
    Set db = CurrentDb
    Set rs = db.OpenRecordset("T_Agendamento", dbOpenDynaset)
    If rs.Fields("Agenda").Type = dbText Then
        If Len(assunto) > rs.Fields("Agenda").Size Then Err.Raise vbObjectError + 1011, , "Assunto excede o tamanho permitido na tabela."
    End If
    Set ws = DBEngine.Workspaces(0)
    ws.BeginTrans
    emTransacao = True
    rs.AddNew
    rs.Fields("Agenda").Value = assunto
    rs.Fields("Data").Value = dataEvento
    If Len(dataFimTxt) > 0 Then
        rs.Fields("DataFimEvento").Value = dataFim
    Else
        rs.Fields("DataFimEvento").Value = Null
    End If
    rs.Fields("Hora").Value = hInicio
    rs.Fields("HoraTermino").Value = hFim
    rs.Fields("Data da Informação").Value = Date
    rs.Fields("Login").Value = Left$(Environ$("USERNAME"), 255)
    rs.Fields("Lida").Value = False
    rs.Update
    rs.Bookmark = rs.LastModified
    novoID = CLng(rs.Fields("ID").Value)
    rs.Close: Set rs = Nothing
    Set ver = db.OpenRecordset("SELECT ID FROM T_Agendamento WHERE ID=" & CStr(novoID), dbOpenSnapshot)
    If ver.EOF Then Err.Raise vbObjectError + 1007, , "A gravacao nao foi confirmada na tabela. Consulte o ID antes de repetir."
    ver.Close: Set ver = Nothing
    ws.CommitTrans
    emTransacao = False
    mAviso = "Compromisso gravado e confirmado. ID " & CStr(novoID) & "."
    AgendaSISDENF_Gravar = True
Saida:
    On Error Resume Next
    If Not ver Is Nothing Then ver.Close
    If Not rs Is Nothing Then rs.Close
    Set db = Nothing: Set rs = Nothing: Set ver = Nothing: Set ws = Nothing
    Exit Function
Falha:
    mAvisoErro = True
    mAviso = "CADASTRO NAO CONFIRMADO: " & Err.Number & " - " & Err.Description
    On Error Resume Next
    If emTransacao Then ws.Rollback
    On Error GoTo 0
    GoTo Saida
End Function

Private Function AgHoraValida(ByVal s As String, ByRef t As Date) As Boolean
    On Error GoTo Invalida
    Dim p() As String, h As Long, m As Long
    p = Split(Trim$(s), ":")
    If UBound(p) <> 1 Then Exit Function
    If Len(p(0)) < 1 Or Len(p(0)) > 2 Or Len(p(1)) <> 2 Then Exit Function
    If Not (p(0) Like String$(Len(p(0)), "#")) Or Not (p(1) Like "##") Then Exit Function
    h = CLng(p(0)): m = CLng(p(1))
    If h > 23 Or h < 0 Or m > 59 Or m < 0 Then Exit Function
    t = TimeSerial(h, m, 0): AgHoraValida = True
Invalida:
End Function

Private Function AgGet(ByVal qs As String, ByVal k As String) As String
    Dim parts() As String, i As Long, a As String, v As String
    parts = Split(qs, "&")
    For i = LBound(parts) To UBound(parts)
        a = parts(i)
        If Left$(a, Len(k) + 1) = k & "=" Then
            v = Mid$(a, Len(k) + 2)
            AgGet = AgUnescape(v)
            Exit Function
        End If
    Next
End Function

Private Function AgUnescape(ByVal s As String) As String
    Dim i As Long, h As String, r As String
    i = 1
    Do While i <= Len(s)
        If Mid$(s, i, 2) = "%u" And i + 5 <= Len(s) Then
            h = Mid$(s, i + 2, 4)
            If h Like "[0-9A-Fa-f][0-9A-Fa-f][0-9A-Fa-f][0-9A-Fa-f]" Then
                r = r & ChrW$(CLng("&H" & h)): i = i + 6
            Else
                r = r & Mid$(s, i, 1): i = i + 1
            End If
        ElseIf Mid$(s, i, 1) = "%" And i + 2 <= Len(s) Then
            h = Mid$(s, i + 1, 2)
            If h Like "[0-9A-Fa-f][0-9A-Fa-f]" Then
                r = r & ChrW$(CLng("&H" & h)): i = i + 3
            Else
                r = r & "%": i = i + 1
            End If
        Else
            r = r & Mid$(s, i, 1): i = i + 1
        End If
    Loop
    AgUnescape = Left$(r, 750)
End Function

Private Function AgSafe(ByVal s As String) As String
    AgSafe = Replace(Replace(Replace(Replace(Replace(s, "&", "&amp;"), "<", "&lt;"), ">", "&gt;"), Chr$(34), "&quot;"), "'", "&#39;")
End Function

Private Function AgDate(ByVal v As Variant) As String
    If Not IsNull(v) Then If IsDate(v) Then AgDate = Format$(CDate(v), "dd/mm/yyyy")
End Function
Private Function AgTime(ByVal v As Variant) As String
    AgTime = "--:--"
    If Not IsNull(v) Then If IsDate(v) Then AgTime = Format$(CDate(v), "hh:nn")
End Function
Private Function AgCatName(ByVal c As String) As String
    Select Case c
    Case "00": AgCatName = "Lembrete"
    Case "01": AgCatName = "Reuniões"
    Case "02": AgCatName = "Urgente"
    Case "03": AgCatName = "Retorno Servidor"
    Case "04": AgCatName = "EMS_NUP Ouvidoria"
    Case Else: AgCatName = "Outros"
    End Select
End Function

End Function

Private Function AgNomeCategoria(ByVal codigo As String) As String
    Dim rs As DAO.Recordset
    If Len(codigo) <> 2 Or Not codigo Like "##" Then
        AgNomeCategoria = "Sem categoria"
        Exit Function
    End If
    Set rs = CurrentDb.OpenRecordset("SELECT [Descricao] FROM [T_AgendaCategorias] WHERE [Codigo]='" & codigo & "'", dbOpenSnapshot)
    If Not rs.EOF Then AgNomeCategoria = Nz(rs!descricao, "") Else AgNomeCategoria = codigo
    rs.Close
End Function

Private Sub AgEditarCategoria(ByVal codigo As String, ByVal descricao As String, _
                              ByVal Cor As String, ByVal ativo As String)
    Dim db As DAO.Database, rs As DAO.Recordset
    Dim i As Long, ch As String
    descricao = Trim$(descricao)
    Cor = UCase$(Trim$(Cor))
    If Len(codigo) <> 2 Or Not codigo Like "##" Then _
        Err.Raise vbObjectError + 1851, , "Codigo invalido."
    If Len(descricao) = 0 Or Len(descricao) > 80 Then _
        Err.Raise vbObjectError + 1852, , "Descricao deve ter de 1 a 80 caracteres."
    If Len(Cor) <> 7 Or Left$(Cor, 1) <> "#" Then _
        Err.Raise vbObjectError + 1853, , "Cor invalida (#RRGGBB)."
    For i = 2 To 7
        ch = Mid$(Cor, i, 1)
        If InStr(1, "0123456789ABCDEF", ch, vbBinaryCompare) = 0 Then _
            Err.Raise vbObjectError + 1854, , "Cor hexadecimal invalida."
    Next i
    If ativo <> "0" And ativo <> "1" Then _
        Err.Raise vbObjectError + 1855, , "Estado ativo invalido."
    Set db = CurrentDb
    Set rs = db.OpenRecordset("SELECT [Codigo],[Descricao],[CorHex],[Ativa] " & _
        "FROM [T_AgendaCategorias] WHERE [Codigo]='" & codigo & "'", dbOpenDynaset)
    If rs.EOF Then Err.Raise vbObjectError + 1856, , "Categoria nao encontrada."
    rs.Edit
    rs.Fields("Descricao").Value = descricao
    rs.Fields("CorHex").Value = Cor
    rs.Fields("Ativa").Value = (ativo = "1")
    rs.Update
    rs.Close
End Sub

Private Function AgCategoriasBotoes() As String
    Dim rs As DAO.Recordset, b As String, c As String, t As String
    Dim Cor As String, ativ As Boolean, texto As String
    Set rs = CurrentDb.OpenRecordset("SELECT [Codigo],[Descricao],[CorHex],[Ativa] " & _
        "FROM [T_AgendaCategorias] ORDER BY [Codigo]", dbOpenSnapshot)
    Do Until rs.EOF
        c = Nz(rs!codigo, ""): t = Nz(rs!descricao, "")
        Cor = Nz(rs!corHex, "#4477AA"): ativ = Nz(rs!Ativa, False)
        If Len(c) = 2 And c Like "##" Then
            texto = c & " - " & AgSafe(t)
            b = b & "<div class='catRow'>"
            If ativ Then
                b = b & "<button type='button' onclick=""cat('" & c & "')"">" & texto & "</button>"
            Else
                b = b & "<span class='catInactive'>" & texto & " (inativa)</span>"
            End If
            b = b & "<button class='catEdit' type='button' title='Editar categoria' " & _
                "data-code='" & AgSafe(c) & "' data-name='" & AgSafe(t) & _
                "' data-color='" & AgSafe(Cor) & "' data-active='" & _
                IIf(ativ, "1", "0") & "' onclick='categoriaEditar(this)'>&#9998;</button></div>"
        End If
        rs.MoveNext
    Loop
    rs.Close
    AgCategoriasBotoes = b
End Function

Private Function AgCategoriasOpcoes() As String
    Dim rs As DAO.Recordset, b As String, c As String, t As String
    Set rs = CurrentDb.OpenRecordset("SELECT [Codigo],[Descricao] FROM [T_AgendaCategorias] WHERE [Ativa]=True ORDER BY [Codigo]", dbOpenSnapshot)
    Do Until rs.EOF
        c = Nz(rs!codigo, ""): t = Nz(rs!descricao, "")
        If Len(c) = 2 And c Like "##" Then b = b & "<option value='" & c & "'>" & c & " - " & AgSafe(t) & "</option>"
        rs.MoveNext
    Loop
    rs.Close
    AgCategoriasOpcoes = b
End Function

Private Function AgBuscaLikeAcentos(ByVal texto As String) As String
    Dim i As Long, ch As String, r As String
    ' Sintaxe LIKE ANSI-89 do Access: * e classes [abc].
    ' Nao usar Replace em classes prontas para nao corromper colchetes.
    For i = 1 To Len(texto)
        ch = Mid$(texto, i, 1)
        Select Case LCase$(ch)
            Case "a", "á", "à", "â", "ã", "ä"
                r = r & "[aáàâãäAÁÀÂÃÄ]"
            Case "e", "é", "è", "ê", "ë"
                r = r & "[eéèêëEÉÈÊË]"
            Case "i", "í", "ì", "î", "ï"
                r = r & "[iíìîïIÍÌÎÏ]"
            Case "o", "ó", "ò", "ô", "õ", "ö"
                r = r & "[oóòôõöOÓÒÔÕÖ]"
            Case "u", "ú", "ù", "û", "ü"
                r = r & "[uúùûüUÚÙÛÜ]"
            Case "c", "ç"
                r = r & "[cçCÇ]"
            Case "n", "ñ"
                r = r & "[nñNÑ]"
            Case "*"
                r = r & "[*]"
            Case "?"
                r = r & "[?]"
            Case "#"
                r = r & "[#]"
            Case "["
                r = r & "[[]"
            Case "]"
                ' Colchete de fechamento isolado e literal no LIKE Access.
                r = r & "]"
            Case "'"
                r = r & "''"
            Case Else
                r = r & ch
        End Select
    Next i
    AgBuscaLikeAcentos = r
End Function

    s = Replace(s, "%%Q%%", AgSafe(mQ)): s = Replace(s, "%%DATA%%", AgSafe(mData))
    s = Replace(s, "%%QJS%%", AgJS(mQ)): s = Replace(s, "%%DATAJS%%", AgJS(mData))
    s = Replace(s, "%%CAT%%", AgSafe(mCat)): s = Replace(s, "%%TIPO%%", AgSafe(mTipo)): s = Replace(s, "%%PERIODO%%", AgSafe(mPeriodo))
    'Mark of the Web: identifica HTML como zona Internet e evita o
    'alerta de conteudo ativo de arquivo local no controle Shell.Explorer.2.
    'Nao altera registro, Trusted Sites ou configuracoes globais do Windows.
    'O marcador deve ser a PRIMEIRA sequencia de bytes do arquivo, sem BOM.
    s = "<!-- saved from url=(0014)about:internet -->" & vbCrLf & s
    mArquivoIndice = 1 - mArquivoIndice
    If Len(mPastaTemporaria) = 0 Then
        Dim fso As Object
        Set fso = CreateObject("Scripting.FileSystemObject")
        mPastaTemporaria = fso.BuildPath(Environ$("TEMP"), fso.GetTempName)
        fso.CreateFolder mPastaTemporaria
        Set fso = Nothing
    End If
    path = mPastaTemporaria & "\Agenda_" & CStr(mArquivoIndice) & ".html"
    Set st = CreateObject("ADODB.Stream")
    st.Type = 2: st.charset = "utf-8": st.Open
    st.WriteText s
    'ADODB.Stream texto UTF-8 inclui BOM EF BB BF: removemos antes de salvar
    'para assegurar que o comentario MOTW apareca no inicio do arquivo.
    st.Position = 0
    st.Type = 1
    Set saidaHTML = CreateObject("ADODB.Stream")
    saidaHTML.Type = 1: saidaHTML.Open
    st.Position = 3
    saidaHTML.Write st.Read
    saidaHTML.SaveToFile path, 2
    saidaHTML.Close: Set saidaHTML = Nothing
    st.Close
    AgendaSISDENF_GerarHTML = path
Saida:
    On Error Resume Next
    If Not rs Is Nothing Then rs.Close
    If Not st Is Nothing Then st.Close
    If Not saidaHTML Is Nothing Then saidaHTML.Close
    Set saidaHTML = Nothing
    Set rs = Nothing: Set db = Nothing: Set st = Nothing
    Exit Function
Falha:
    er = CStr(Err.Number) & " - " & Err.Description
    Resume SaidaErro
SaidaErro:
    On Error Resume Next
    If Not rs Is Nothing Then rs.Close
    If Not st Is Nothing Then st.Close
    If Not saidaHTML Is Nothing Then saidaHTML.Close
    On Error GoTo 0
    Err.Raise vbObjectError + 910, "AgendaSISDENF_GerarHTML", er
End Function

Private Function AgJS(ByVal s As String) As String
    Dim t As String
    t = Replace(s, "\", "\\")
    t = Replace(t, "'", "\'")
    t = Replace(t, vbCr, "\r")
    t = Replace(t, vbLf, "\n")
    t = Replace(t, "<", "\u003C")
    t = Replace(t, ChrW$(&H2028), "\u2028")
    t = Replace(t, ChrW$(&H2029), "\u2029")
    AgJS = t
End Function

Private Function AgParseDate(ByVal s As String, ByRef d As Date) As Boolean
    On Error GoTo Nao
    Dim p() As String, y As Long, m As Long, diaNum As Long
    p = Split(s, "/")
    If UBound(p) <> 2 Then Exit Function
    If Len(p(2)) <> 4 Then Exit Function
    If Len(p(0)) < 1 Or Len(p(0)) > 2 Or Len(p(1)) < 1 Or Len(p(1)) > 2 Then Exit Function
    If Not (p(0) Like String$(Len(p(0)), "#")) Or Not (p(1) Like String$(Len(p(1)), "#")) Or Not (p(2) Like "####") Then Exit Function
    diaNum = CLng(p(0)): m = CLng(p(1)): y = CLng(p(2))
    If y < 100 Or y > 9998 Then Exit Function
    d = DateSerial(y, m, diaNum)
    AgParseDate = (Year(d) = y And Month(d) = m And VBA.day(d) = diaNum)
    Exit Function
Nao:
    AgParseDate = False
End Function




Private Function AgIDValido(ByVal texto As String) As Long
    If Len(texto) < 1 Or Len(texto) > 10 Then Err.Raise vbObjectError + 1932, , "ID invalido."
    If Not (texto Like String$(Len(texto), "#")) Then Err.Raise vbObjectError + 1932, , "ID deve conter apenas algarismos."
    If CDbl(texto) > 2147483647# Then Err.Raise vbObjectError + 1932, , "ID fora do limite."
    AgIDValido = CLng(texto)
    If AgIDValido < 1 Then Err.Raise vbObjectError + 1932, , "ID deve ser positivo."
End Function

Private Function AgCorCategoria(ByVal codigo As String) As String
    On Error GoTo Saida
    Dim rs As DAO.Recordset, cor As String, i As Long
    AgCorCategoria = "#4796E8"
    If Len(codigo) <> 2 Or Not (codigo Like "##") Then Exit Function
    Set rs = CurrentDb.OpenRecordset("SELECT CorHex FROM T_AgendaCategorias WHERE Codigo='" & codigo & "'", dbOpenSnapshot)
    If rs.EOF Then GoTo Saida
    cor = UCase$(Nz(rs!CorHex, ""))
    If Len(cor) <> 7 Or Left$(cor, 1) <> "#" Then GoTo Saida
    For i = 2 To 7
        If InStr(1, "0123456789ABCDEF", Mid$(cor, i, 1), vbBinaryCompare) = 0 Then GoTo Saida
    Next i
    AgCorCategoria = cor
Saida:
    On Error Resume Next
    If Not rs Is Nothing Then rs.Close
End Function

Private Function AgAssuntoExibicao(ByVal texto As String) As String
    texto = Trim$(texto)
    If Len(texto) >= 3 And Left$(texto, 2) Like "##" Then
        If Mid$(texto, 3, 3) = " - " Then
            texto = Trim$(Mid$(texto, 6))
        ElseIf Mid$(texto, 3, 1) = " " Then
            texto = Trim$(Mid$(texto, 4))
        End If
    End If
    AgAssuntoExibicao = texto
End Function
